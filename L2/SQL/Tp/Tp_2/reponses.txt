Exo 2.3 :
SELECT lign.id_ligne, comm.id_commande, cli.nom, cli.statut_fidelite, prod.nom, lign.remise_pct, gril.remise_max_pct
FROM techstor_clients 		cli
JOIN techstor_commandes 	comm USING(id_client)
JOIN techstor_lignescommande 	lign USING(id_commande)
JOIN techstor_produits 		prod USING(id_produit)
JOIN techstor_grillefidelite 	gril USING(categorie)
WHERE cli.statut_fidelite = gril.statut_fidelite
AND lign.remise_pct > gril.remise_max_pct
ORDER BY lign.id_ligne; 

	9 lignes ne respectent pas la grille de remise.

Exo 3.1 :
SELECT DISTINCT prod.categorie, SUM(lign.quantite*((prod.prix_unitaire*(1-lign.remise_pct/100))-prod.prix_achat))
FROM techstor_produits prod
NATURAL JOIN techstor_lignescommande lign
GROUP BY prod.categorie
HAVING SUM(lign.quantite*((prod.prix_unitaire*(1-lign.remise_pct/100))-prod.prix_achat)) > 1000;

	Il n'y a que la categorie Moniteur qui dépasse.

Exo 3.3 :
SELECT prod.categorie, COUNT(sav.id_retour)
FROM techstor_produits 		prod
NATURAL JOIN techstor_sav 	sav
GROUP BY prod.categorie
HAVING COUNT(sav.id_retour) >= ALL(
	SELECT COUNT(id_retour)
	FROM techstor_produits
	NATURAL JOIN techstor_sav
	GROUP BY categorie
);

	La categorie avec le plus de retour est Moniteur avec 6 retours
