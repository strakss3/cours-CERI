Exo 2.1 :
    SELECT dessta, scusta
    FROM statue sta
    JOIN musee mus USING(codmus)
    WHERE nommus='Musée du Louvre';


Exo 2.2 :
    SELECT numsta, codbat
    FROM statue sta
    JOIN batiment bat
    ON sta.codbat = bat.codbat
    AND sta.codsit = bat.codsit
    AND sta.codvil_s = bat.codvil
    WHERE sta.datsta < bat.datbat;


Exo 2.3 :
    SELECT mus.codmus
    FROM musee mus
    LEFT JOIN statue sta
    ON sta.codvil_m = mus.codvil
    AND sta.codmus = mus.codmus
    WHERE numsta IS NULL;


Exo 3.1.A :
    SELECT ouvr.titouv, COUNT(numsta)
    FROM info_statue inf
    JOIN ouvrage ouvr USING(numouv)
    GROUP BY ouvr.numouv, ouvr.titouv;


Exo 3.1.B :
    SELECT ouvr.titouv, sta.dessta
    FROM ouvrage ouvr
    JOIN info_statue info USING(numouv)
    JOIN statue sta USING(numsta)
    WHERE sta.dessta ILIKE '%amour%';


Exo 3.2.A :
    SELECT sta.dessta
    FROM statue sta
    JOIN info_statue info USING(numsta)
    GROUP BY sta.numsta, sta.dessta 
    HAVING COUNT(sta) > 1;


Exo 3.2.B :
    SELECT sta.dessta
    FROM co_auteur co 
    JOIN ouvrage ouvr USING(numouv) 
    JOIN info_statue info USING(numouv)
    JOIN statue sta USING(numsta)
    GROUP BY numouv, dessta
    HAVING COUNT(aut) > 1;


Exo 3.3.A :
    SELECT anom, COUNT(numouv)
    FROM ville vil
    JOIN statue sta
    ON vil.codvil = sta.codvil_o
    JOIN info_statue info USING(numsta)
    GROUP BY vil.codvil;


Exo 3.3.B :
    SELECT anom, COUNT(numouv)
    FROM ville vil
    LEFT JOIN statue sta 
    ON vil.codvil = sta.codvil_o 
    LEFT JOIN info_statue info USING(numsta)
    GROUP BY vil.codvil;


Exo 3.4 :
    SELECT DISTINCT nomedi
    FROM editeur editeur
    LEFT JOIN ouvrage ouvr USING(numedi)
    LEFT JOIN info_statue info USING(numouv)
    WHERE info.numsta IS NULL;


Exo 4.1 :
    SELECT nomaut, COUNT(numouv)
    FROM auteur
    JOIN co_auteur co USING(aut)
    GROUP BY co.aut, nomaut
    HAVING COUNT(numouv) >= ALL (
        SELECT COUNT(numouv)
        FROM co_auteur co
        GROUP BY aut
    );


Exo 4.2 :
    SELECT titouv COUNT(numsta)
    FROM info_statue info
    JOIN ouvrage USING(numouv)
    GROUP BY numouv, titouv
    HAVING COUNT(DISTINCT numsta) >= ALL(
        SELECT COUNT(DISTINCT numsta)
        FROM info_statue
        GROUP BY numouv
    );
