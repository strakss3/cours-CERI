1.1 :
	SELECT ven.nom, cli.nom, cli.ville, com.date_commande, pro.nom, lig.quantite
	FROM techstor_clients cli
	JOIN techstor_commandes com USING(id_client)
	JOIN techstor_vendeur ven USING(id_vendeur)
	JOIN techstor_lignescommande lig USING(id_commande)
	JOIN techstor_produits pro USING(id_produit)
	WHERE cli.ville='Lyon';

1.2 :
	SELECT lign.quantite*lign.prix_unitaire
	FROM techstor_lignescommande lign
	JOIN techstor_commandes com USING(id_commande);

1.3 :
	SELECT com.id_commande, vend.nom, vend.date_embauche
	FROM techstor_commandes com
	JOIN techstor_vendeur vend USING(id_vendeur)
	JOIN techstor_clients cli USING(id_client)
	WHERE cli.ville IN ('Lyon', 'Marseille', 'Paris') AND vend.date_embauche>='2026-01-01';

2.1 :
	SELECT cli.ville, prod.categorie, SUM(com.montant_total)
	FROM techstor_clients cli
	JOIN techstor_commandes com USING(id_client)
	JOIN techstor_lignescommande lign USING(id_commande)
	JOIN techstor_produits prod USING(id_produit)
	GROUP BY cli.ville, prod.categorie;

2.2 :
	SELECT 
		vend.id_vendeur,
		EXTRACT(YEAR FROM comm.date_commande) AS annee,
		COUNT(id_commande),
		SUM(comm.montant_total)
	FROM techstor_commandes comm
	JOIN techstor_vendeur vend USING(id_vendeur)
	GROUP BY vend.id_vendeur, annee
	ORDER BY vend.id_vendeur, annee;

2.3 :
	SELECT
		vend.id_vendeur,
		cli.id_client,
		COUNT(comm.id_commande),
		AVG(comm.id_commande)
        FROM techstor_clients cli
        JOIN techstor_commandes comm USING(id_client)
        JOIN techstor_vendeur vend USING(id_vendeur)
        GROUP BY vend.id_vendeur, cli.id_client
        ORDER BY vend.id_vendeur, cli.id_client;

3.1 :
	SELECT 
		vend.id_vendeur, 
		cli.ville, 
		COUNT(comm.id_commande)
	FROM techstor_clients cli
	JOIN techstor_commandes comm USING(id_client)
	JOIN techstor_vendeur vend USING(id_vendeur)
	GROUP BY vend.id_vendeur, cli.ville
	HAVING COUNT(comm.id_commande) > 3
	ORDER BY vend.id_vendeur, cli.ville;

3.2 :
	SELECT
		prod.categorie,
		EXTRACT(YEAR FROM comm.date_commande) AS annee,
		SUM(lign.quantite*lign.prix_unitaire) AS montant
	FROM techstor_commandes comm
	JOIN techstor_lignescommande lign USING(id_commande)
	JOIN techstor_produits prod USING(id_produit)
	GROUP BY prod.categorie, annee
	HAVING SUM(lign.quantite*lign.prix_unitaire) > 1500
	AND EXTRACT(YEAR FROM comm.date_commande) = 2026;

3.3 :
	SELECT
		prenom,
		nom,
		COUNT(id_commande),
		AVG(montant_total)
	FROM techstor_vendeur vend
	JOIN techstor_commandes comm USING(id_vendeur)
	GROUP BY id_vendeur
	HAVING COUNT(id_commande)>=2
	AND AVG(montant_total)>400;
